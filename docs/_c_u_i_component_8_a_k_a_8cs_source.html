123 /// <summary>
124 /// Returns memorized component by name.
125 /// You can chain names with . and get parent with ^
126 /// </summary>
127 public virtual CUIComponent Get(string name)
128 {
129 if (name == null || name == "") return null;
130 name = name.Trim();
131
132 if (NamedComponents.ContainsKey(name)) return NamedComponents[name];
133
134 CUIComponent component = this;
135 string[] commands = name.Split('.');
136
137 foreach (string c in commands)
138 {
139 if (c == GetParentLitteral)
140 {
141 if (component.Parent == null)
142 {
143 CUI.Logger.Warning($"Failed to Get [{name}] from [{this}], [{component}] has no parent");
144 return null;
145 }
146
147 component = component.Parent;
148 }
149 else
150 {
151 if (!component.NamedComponents.ContainsKey(c))
152 {
153 CUI.Logger.Warning($"Failed to Get [{name}] from [{this}], [{component}] has no [{c}]");
154 return null;
155 }
156
157 component = component.NamedComponents[c];
158 }
159 }
160
161 return component;
162 }
163 public T Get<T>(string name) where T : CUIComponent => (T)Get(name);
