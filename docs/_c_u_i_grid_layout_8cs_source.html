54 public override void UpdateChildren()
55 {
56 if (Parent is null) return;
57 if (!RequireChildrenUpdate) return;
58
59 if (Parent.RowSizes.Count == 0 || Parent.ColumnSizes.Count == 0)
60 {
61 base.UpdateChildren();
62 return;
63 }
64
65
66 List<CUISegment> RealRowSizes = new();
67 List<CUISegment> RealColumnSizes = new();
68
69 float y = Parent.ChildrenRect.Top;
70 float fry = Parent.ChildrenRect.Height / Parent.RowSizes.Where(l => l.Fraction.HasValue).Sum(l => l.Fraction.Value);
71 foreach (Line line in Parent.RowSizes)
72 {
73 float size = 0;
74 if (line.Fraction.HasValue) size = fry * line.Fraction.Value;
75 if (line.RelativeSize.HasValue) size = Parent.ChildrenRect.Height * line.RelativeSize.Value;
76 if (line.AbsoluteSize.HasValue) size = line.AbsoluteSize.Value;
77
78 RealRowSizes.Add(new CUISegment() { Left = y, Width = size });
79
80 y += size;
81 }
82
83
84 float x = Parent.ChildrenRect.Left;
85 float frx = Parent.ChildrenRect.Width / Parent.ColumnSizes.Where(l => l.Fraction.HasValue).Sum(l => l.Fraction.Value);
86 foreach (Line line in Parent.ColumnSizes)
87 {
88 float size = 0;
89 if (line.Fraction.HasValue) size = frx * line.Fraction.Value;
90 if (line.RelativeSize.HasValue) size = Parent.ChildrenRect.Width * line.RelativeSize.Value;
91 if (line.AbsoluteSize.HasValue) size = line.AbsoluteSize.Value;
92
93 RealColumnSizes.Add(new CUISegment() { Left = x, Width = size });
94
95 x += size;
96 }
97
98
99
100 foreach (Layout.Child c in Parent.Children)
101 {
102 int i = Math.Clamp(c.GridColumn - 1, 0, RealColumnSizes.Count - 1);
103 int j = Math.Clamp(c.GridRow - 1, 0, RealRowSizes.Count - 1);
104
105 c.OuterRect = new CUIRect(
106 RealColumnSizes[i].Left,
107 RealRowSizes[j].Left,
108 RealColumnSizes[i].Width,
109 RealRowSizes[j].Width
110 );
111 }
112
113 base.UpdateChildren();
114 }
