78 /// <param name="point">([0..1], [0..1])</param>
79 public bool IsPointOnTransparentPixel(Vector2 point)
80 {
81 Rectangle SourceRect = SourceRectangle.HasValue ? SourceRectangle.Value : Texture.Bounds;
82
83 int textureX = (int)Math.Round(SourceRect.X + point.X * SourceRect.Width);
84 int textureY = (int)Math.Round(SourceRect.Y + point.Y * SourceRect.Height);
85
86 if (textureX < SourceRect.X || (SourceRect.X + SourceRect.Width - 1) < textureX) return true;
87 if (textureY < SourceRect.Y || (SourceRect.Y + SourceRect.Height - 1) < textureY) return true;
88
89 Color cl = Data[textureY * Texture.Width + textureX];
90
91 return cl.A == 0;
92 }
