48 public void Run(string name)
49 {
50 if (!Repo.Tests.ContainsKey(name))
51 {
52 Logger.Warning($"Can't find snapshot test: [{name}]");
53 return;
54 }
55
56 ModStorage.Set("CUITest", name);
57
58 if (!Chamber.IsSetup) Chamber.Setup();
59
60 CurrentTest = Repo.Tests[name];
61 CurrentSnapshot = Runner.Run(CurrentTest);
62 ComponentSnapshot stored = GetStoredSnapshot(CurrentTest);
63
64 if (stored is null)
65 {
66 Accept(CurrentTest, CurrentSnapshot);
67 Events.Raise(new Event("passed", CurrentTest));
68 return;
69 }
70
71 if (ComponentSnapshot.AreEqual(stored, CurrentSnapshot))
72 {
73 Events.Raise(new Event("passed", CurrentTest));
74 }
75 else
76 {
77 Events.Raise(new Event("failed", CurrentTest, ComponentSnapshot.CreateDiffString(CurrentSnapshot, stored)));
78 }
79 }
